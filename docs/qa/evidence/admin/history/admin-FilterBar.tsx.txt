"use client";
import { useRouter, useSearchParams } from "next/navigation";
import { FormEvent } from "react";
export type FilterDef = { name:string; label:string; options?:readonly {value:string;label:string}[]; placeholder?:string };
export default function FilterBar({filters,defaults={}}:{filters:readonly FilterDef[];defaults?:Record<string,string>}) { const router=useRouter(); const search=useSearchParams(); function submit(e:FormEvent<HTMLFormElement>){e.preventDefault();const data=new FormData(e.currentTarget);const q=new URLSearchParams();filters.forEach(f=>{const v=String(data.get(f.name)??"").trim();if(v)q.set(f.name,v)});q.set("page","0");router.push(`?${q}`)} const clear=()=>{const q=new URLSearchParams(defaults);router.push(q.toString()?`?${q}`:"?")};return <form onSubmit={submit} className="mb-6 grid gap-3 border border-rule bg-paper p-4 sm:grid-cols-2 lg:grid-cols-5">{filters.map(f=><label key={f.name} className="label label-ink">{f.label}{f.options?<select className="form-input" name={f.name} defaultValue={search.get(f.name)??defaults[f.name]??""}><option value="">All</option>{f.options.map(o=><option key={o.value} value={o.value}>{o.label}</option>)}</select>:<input className="form-input" name={f.name} defaultValue={search.get(f.name)??defaults[f.name]??""} placeholder={f.placeholder}/>}</label>)}<div className="flex items-end gap-2"><button className="btn" type="submit">Apply</button><button className="btn" type="button" onClick={clear}>Clear</button></div></form> }
