 public static void main(String[] args)throws Exception{

   Repository configuring=doAnswer(inv->"barrier answer").when(repository);
   pool.submit(()->repository.findById()).get(5,TimeUnit.SECONDS);

  }finally{pool.shutdownNow();}
 }
}
